cmake_minimum_required(VERSION 3.16)
project(ProcWorld CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(PWG_SANITIZE "Build with AddressSanitizer and UBSan" OFF)
if(PWG_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

find_package(ZLIB REQUIRED)
add_library(fastnoise STATIC third_party/FastNoise.cpp)
target_include_directories(fastnoise PUBLIC third_party)
target_compile_options(fastnoise PRIVATE -w)

add_executable(ProcWorld main.cpp)
target_include_directories(ProcWorld PRIVATE include)
target_link_libraries(ProcWorld PRIVATE fastnoise ZLIB::ZLIB)
find_package(SFML 3 COMPONENTS Graphics QUIET)
if(SFML_FOUND)
  target_link_libraries(ProcWorld PRIVATE SFML::Graphics)
  message(STATUS "SFML 3 found: building the viewer")
else()
  find_package(SFML 2.5 COMPONENTS graphics window system QUIET)
  if(SFML_FOUND)
    target_link_libraries(ProcWorld PRIVATE sfml-graphics sfml-window sfml-system)
    message(STATUS "SFML 2 found: building the viewer")
  else()
    target_compile_definitions(ProcWorld PRIVATE PWG_NO_SFML)
    message(STATUS "SFML not found: building without the viewer (--export / --stats only)")
  endif()
endif()

enable_testing()
find_path(DOCTEST_INCLUDE doctest/doctest.h)
if(DOCTEST_INCLUDE)
  add_executable(pwg_tests tests/test_world.cpp)
  target_include_directories(pwg_tests PRIVATE include ${DOCTEST_INCLUDE})
  target_link_libraries(pwg_tests PRIVATE fastnoise ZLIB::ZLIB)
  add_test(NAME pwg_tests COMMAND pwg_tests)
endif()
